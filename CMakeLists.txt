# CMake configuration for the in-tree VisibilityContours Stellarium plugin.
SET(VISIBILITYCONTOURS_VERSION "0.1.0")
ADD_DEFINITIONS(-DBUILD_DYNAMIC_PLUGIN=0)
ADD_DEFINITIONS(-DVISIBILITYCONTOURS_PLUGIN_VERSION="${VISIBILITYCONTOURS_VERSION}")
ADD_DEFINITIONS(-DVISIBILITYCONTOURS_PLUGIN_LICENSE="GNU GPLv2 or later")

ADD_SUBDIRECTORY(src)

IF(APPLE)
    SET(CMAKE_INSTALL_PREFIX $ENV{HOME}/Library/Application\ Support/Stellarium)
ELSE(APPLE)
    SET(CMAKE_INSTALL_PREFIX $ENV{HOME}/.stellarium)
ENDIF(APPLE)

INSTALL(FILES DESTINATION "modules/VisibilityContours")
